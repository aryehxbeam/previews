    How persistence works

    const KEY = "todo.items.v1";

// Load on startup, falling back to an empty list
function load() {
  try {
    return JSON.parse(localStorage.getItem(KEY)) || [];
  } catch {
    return [];
  }
}

    State flow

    flowchart TD
    A([Page load]) --> B[load from localStorage]
    B --> C{any saved todos?}
    C -->|yes| D[render list]
    C -->|no| E[show empty state]
    D --> F[user action]
    E --> F
    F -->|add| G[push new todo]
    F -->|toggle| H[flip done flag]
    F -->|delete| I[remove todo]
    F -->|clear done| J[drop completed]
    G --> K[(save to localStorage)]
    H --> K
    I --> K
    J --> K
    K --> D
